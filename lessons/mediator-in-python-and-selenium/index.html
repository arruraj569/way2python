Mediator in Python and Selenium

Behavioral patterns · Complete program, tricky exercise, and interview answers

First time writing Python?Start from zero →Read the punctuation guide ↗

The idea

A mediator coordinates collaborating objects so they send requests to it instead of directly managing each other. Here a save request coordinates validation and notification.

Behavioral patterns · Situational

Use it in daily work.

Coordinate a daily account journey where profile changes affect notifications and reporting, while keeping the component classes independent.

Know the tradeoff.

A mediator can become a large object containing every rule. Split by cohesive workflows. Distinguish coordination from a facade’s simpler client entry point.

class NameField:
    def __init__(self, value):
        self.value = value

class SaveButton:
    def __init__(self, mediator):
        self.mediator = mediator
    def click(self):
        return self.mediator.notify(self, "save")

class ProfileMediator:
    def __init__(self, field):
        self.field = field
    def notify(self, sender, event):
        if event != "save":
            raise ValueError("unknown event")
        return "saved " + self.field.value if self.field.value.strip() else "name required"

field = NameField("Mira")
button = SaveButton(ProfileMediator(field))
print(button.click())
field.value = " "
print(button.click())

"""Run against the local Way2Python lab with Selenium and installed Chrome."""
import os
from contextlib import contextmanager
from copy import deepcopy
from dataclasses import dataclass
from urllib.parse import urlsplit
from selenium import webdriver
from selenium.webdriver.common.by import By
from selenium.webdriver.support.ui import WebDriverWait, Select
from selenium.webdriver.support import expected_conditions as EC
from selenium.common.exceptions import TimeoutException

BASE_URL = os.getenv("WAY2PYTHON_URL", "http://localhost:8080/way2python").rstrip("/")

def create_driver():
    options = webdriver.ChromeOptions()
    options.add_argument("--headless=new")
    return webdriver.Chrome(options=options)

@contextmanager
def browser_session(factory=create_driver):
    driver = factory()
    try:
        driver.get(BASE_URL + "/practice.html")
        yield driver
    finally:
        driver.quit()

def control(driver, selector):
    return WebDriverWait(driver, 5).until(
        EC.element_to_be_clickable((By.CSS_SELECTOR, selector)))

def fill(driver, selector, value):
    field = control(driver, selector)
    field.clear()
    field.send_keys(value)

def wait_text(driver, selector, expected):
    WebDriverWait(driver, 5).until(
        lambda current: current.find_element(By.CSS_SELECTOR, selector).text == expected)
    return driver.find_element(By.CSS_SELECTOR, selector).text

def login(driver, email="learner@example.test", password="python123"):
    fill(driver, "#email", email)
    fill(driver, "#password", password)
    control(driver, "#login-form button[type=submit]").click()

def save_profile(driver, name, track="selenium"):
    fill(driver, "#display-name", name)
    Select(control(driver, "#track-select")).select_by_value(track)
    control(driver, "#save-profile").click()


class ProfileForm:
    def __init__(self, driver):
        self.driver = driver
    def apply(self, name, track):
        save_profile(self.driver, name, track)
        return wait_text(self.driver, "#profile-status", "Profile saved: " + name + " · Selenium · updates off")

class ProfileMediator:
    def __init__(self, form, notify):
        self.form, self.notify = form, notify
    def request_save(self, name, track):
        if not name.strip():
            raise ValueError("name required")
        result = self.form.apply(name, track)
        self.notify(result)
        return result

def main():
    with browser_session() as driver:
        notifications = []
        mediator = ProfileMediator(ProfileForm(driver), notifications.append)
        result = mediator.request_save("Mira", "selenium")
        assert result == "Profile saved: Mira · Selenium · updates off"
        assert notifications == [result]
    print('Profile mediator coordination passed')

if __name__ == "__main__":
    main()
EXPECTED RESULT WHEN RUN LOCALLY
Profile mediator coordination passed

Mediator versus Observer?

Mediator owns coordination rules among colleagues. Observer broadcasts events to subscribers; it usually does not decide a whole workflow. A mediator may use observers for notifications.

Which common mistake can affect this program or test?
