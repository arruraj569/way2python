Page Object Model in Python and Selenium

Python & Selenium approaches · Complete program, tricky exercise, and interview answers

First time writing Python?Start from zero →Read the punctuation guide ↗

The idea

A page object stores locators and page operations in one class. Tests describe a journey and check its outcome. This first program uses an explicitly labelled driver double: a small Python substitute that records calls, not a browser.

Python & Selenium approaches · Everyday

Use it in daily work.

When a login button or field ID changes, update LoginPage once and keep your daily login smoke tests readable.

Know the tradeoff.

Keep business-result assertions in tests. A page may check that its expected elements are ready. Do not hide assertions or use a giant base page for unrelated features.

COMPLETE PYTHON PROGRAM
class DriverDouble:
    def __init__(self):
        self.fields = {}
    def fill(self, name, value):
        self.fields[name] = value
    def submit(self):
        return self.fields == {"email": "learner@example.test", "password": "python123"}

class LoginPage:
    def __init__(self, driver):
        self.driver = driver
    def login(self, email, password):
        self.driver.fill("email", email)
        self.driver.fill("password", password)
        return self.driver.submit()

page = LoginPage(DriverDouble())
result = page.login("learner@example.test", "python123")
assert result is True
print("Driver double: login passed")
EXPECTED RESULT
Driver double: login passed

"""Run against the local Way2Python lab with Selenium and installed Chrome."""
import os
from contextlib import contextmanager
from copy import deepcopy
from dataclasses import dataclass
from urllib.parse import urlsplit
from selenium import webdriver
from selenium.webdriver.common.by import By
from selenium.webdriver.support.ui import WebDriverWait, Select
from selenium.webdriver.support import expected_conditions as EC
from selenium.common.exceptions import TimeoutException

BASE_URL = os.getenv("WAY2PYTHON_URL", "http://localhost:8080/way2python").rstrip("/")

def create_driver():
    options = webdriver.ChromeOptions()
    options.add_argument("--headless=new")
    return webdriver.Chrome(options=options)

@contextmanager
def browser_session(factory=create_driver):
    driver = factory()
    try:
        driver.get(BASE_URL + "/practice.html")
        yield driver
    finally:
        driver.quit()

def control(driver, selector):
    return WebDriverWait(driver, 5).until(
        EC.element_to_be_clickable((By.CSS_SELECTOR, selector)))

def fill(driver, selector, value):
    field = control(driver, selector)
    field.clear()
    field.send_keys(value)

def wait_text(driver, selector, expected):
    WebDriverWait(driver, 5).until(
        lambda current: current.find_element(By.CSS_SELECTOR, selector).text == expected)
    return driver.find_element(By.CSS_SELECTOR, selector).text

def login(driver, email="learner@example.test", password="python123"):
    fill(driver, "#email", email)
    fill(driver, "#password", password)
    control(driver, "#login-form button[type=submit]").click()

def save_profile(driver, name, track="selenium"):
    fill(driver, "#display-name", name)
    Select(control(driver, "#track-select")).select_by_value(track)
    control(driver, "#save-profile").click()


class LoginPage:
    def __init__(self, driver):
        self.driver = driver
        control(driver, "#email")
    def sign_in(self, email, password):
        login(self.driver, email, password)
    def welcome(self):
        return wait_text(self.driver, "#login-status", "Welcome, learner!")

def main():
    with browser_session() as driver:
        page = LoginPage(driver)
        page.sign_in("learner@example.test", "python123")
        assert page.welcome() == "Welcome, learner!"
    print('Page Object Model login passed')

if __name__ == "__main__":
    main()
EXPECTED RESULT WHEN RUN LOCALLY
Page Object Model login passed

How is a page object different from a test?

The page object knows the page and offers operations such as login. The test chooses data and asserts the result. The live example uses ordinary Python classes and Selenium locators; no Java PageFactory annotation is needed.

Which common mistake can affect this program or test?

Keep business-result assertions in tests. A page may check that its expected elements are ready. Do not hide assertions or use a giant base page for unrelated features.
